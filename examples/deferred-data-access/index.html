deferred-data-access proxy command handler

Every property access, assignment, and method call on a wrapped object becomes a typed command delivered to an async handler. The command chain is only resolved when you await the expression.

const { handle } = DeferredDataAccess;
const { ProxyCommand } = DeferredDataAccess;

const wrap = handle(async (command, context) => {
  const target = await context;

  switch (command.type) {
    case ProxyCommand.GET:
      return target[command.name];

    case ProxyCommand.SET:
      target[command.name] = command.value;
      return command.value;

    case ProxyCommand.METHOD_CALL:
      return target[command.name](
        ...(command.value ?? [])
      );
  }
});

const proxy = wrap(myObject);
const name  = await proxy.user.name; // → handler fires
